    class GameSession:
    """The whole game state: phase machine, current level and its fixed-step simulation.

    A crash restarts the level, as many times as it takes. A test ride from the editor plays one level the same way,
    and goes back to the editor when the level is won or left.
    """

    MAX_STEPS_PER_UPDATE = 30

    def __init__(self, levels: Sequence[Level], config: GameConfig | None = None) -> None:
        """Create a session on the level select screen, with the first level loaded."""
        if not levels:
            msg = "a session needs at least one level"
            raise ValueError(msg)
        self.config = config or GameConfig()
        self.levels = tuple(levels)
        self._test_level: Level | None = None
        self.phase = GamePhase.LEVEL_SELECT
        self.selected_level = 0
        self._physics = ShipPhysics(self.config)
        self._effects = TileEffectApplier(self.config)
        self._accumulator = 0.0
        self._phase_timer = 0.0
        self._load_level(0)

    @property
    def level(self) -> Level:
        """The level being played."""
        return self._test_level or self.levels[self.level_index]

    @property
    def riding(self) -> bool:
        """Whether the level being played is a test ride from the editor."""
        return self._test_level is not None

    def put_level(self, index: int, level: Level) -> None:
        """Replace the level at ``index`` with ``level``, or add it after the last one when ``index`` is past it."""
        levels = list(self.levels)
        if index < len(levels):
            levels[index] = level
        else:
            levels.append(level)
        self.levels = tuple(levels)

    def open_editor(self) -> None:
        """Switch to the level editor, ending any test ride."""
        self._test_level = None
        self.phase = GamePhase.EDITOR

    def ride(self, level: Level) -> list[GameEvent]:
        """Start a test ride of ``level``, from the editor."""
        self._test_level = level
        self._load(level)
        self.phase = GamePhase.PLAYING
        return [GameEvent.LEVEL_STARTED]

    @property
    def progress(self) -> float:
        """How far along the track the ship is, from 0 to 1."""
        return min(1.0, max(0.0, self.ship.position.z / self.level.track.length))

    def open_level_select(self) -> None:
        """Switch to the level selection screen."""
        self._test_level = None
        self.phase = GamePhase.LEVEL_SELECT

    def select_next_level(self) -> None:
        """Select the next level, wrapping around to the first."""
        self.selected_level = (self.selected_level + 1) % len(self.levels)

    def select_previous_level(self) -> None:
        """Select the previous level, wrapping around to the last."""
        self.selected_level = (self.selected_level - 1) % len(self.levels)

    def select_level_column(self, step: int, column_size: int) -> None:
        """Move the selection ``step`` columns sideways, the levels being listed in columns of ``column_size``.

        The selection keeps its row, or takes the last level of a shorter column, and wraps around the columns.
        """
        columns = -(-len(self.levels) // column_size)
        column, row = divmod(self.selected_level, column_size)
        self.selected_level = min((column + step) % columns * column_size + row, len(self.levels) - 1)

    def start(self) -> list[GameEvent]:
        """Start a new game at the selected level."""
        return self._begin_level(self.selected_level)

    def update(self, dt: float, ship_input: ShipInput) -> list[GameEvent]:
        """Advance the session by ``dt`` seconds and return the events that happened."""
        if self.phase is GamePhase.PLAYING:
            return self._simulate(dt, ship_input)
        if self.phase in (GamePhase.DYING, GamePhase.LEVEL_WON):
            self._phase_timer -= dt
            if self._phase_timer <= 0:
                return self._finish_transition()
        return []

    def _simulate(self, dt: float, ship_input: ShipInput) -> list[GameEvent]:
        step = self.config.fixed_step
        self._accumulator = min(self._accumulator + dt, step * self.MAX_STEPS_PER_UPDATE)
        events: list[GameEvent] = []
        while self._accumulator >= step and self.phase is GamePhase.PLAYING:
            self._accumulator -= step
            events.extend(self._step(ship_input, step))
        return events

    def _step(self, ship_input: ShipInput, dt: float) -> list[GameEvent]:
        ship = self.ship
        was_grounded = ship.grounded
        start_z = ship.position.z
        jump = self._physics.apply(ship, ship_input, self.level.gravity, dt)
        events = [] if jump is None else [jump]
        events += self._handle_collisions(
            self._resolver.move(ship, dt, took_off=jump is GameEvent.JUMPED), was_grounded=was_grounded
        )
        if self.phase is not GamePhase.PLAYING:
            return events
        self.fuel.drain((ship.position.z - start_z) * self.config.fuel_per_unit)
        self.oxygen.drain(dt)
        if self.fuel.is_empty:
            return [*events, *self._die(GameEvent.OUT_OF_FUEL)]
        if self.oxygen.is_empty:
            return [*events, *self._die(GameEvent.OUT_OF_OXYGEN)]
        if ship.position.z >= self.level.track.length:
            return [*events, *self._win()]
        return events

    def _handle_collisions(self, reports: list[CollisionReport], *, was_grounded: bool) -> list[GameEvent]:
        events: list[GameEvent] = []
        surface = None
        for report in reports:
            match report.outcome:
                case CollisionOutcome.CRASHED:
                    return [*events, *self._die(GameEvent.CRASHED)]
                case CollisionOutcome.FELL:
                    return [*events, *self._die(GameEvent.FELL)]
                case CollisionOutcome.NUDGED:
                    events.append(GameEvent.NUDGED)
                case CollisionOutcome.BUMPED:
                    events.append(GameEvent.BUMPED)
                case CollisionOutcome.SCRAPED_LEFT:
                    events.append(GameEvent.SCRAPED_LEFT)
                case CollisionOutcome.SCRAPED_RIGHT:
                    events.append(GameEvent.SCRAPED_RIGHT)
                case _:  # CollisionOutcome.LANDED
                    if not was_grounded:
                        events.append(GameEvent.LANDED)
                    if report.tile is not None:
                        events.extend(self._apply_tile(report.tile))
                        surface = report.tile.tile_type
        self.ship.surface = surface
        return events

    def _apply_tile(self, tile: Tile) -> list[GameEvent]:
        event = self._effects.apply(tile, self.ship, self.fuel, self.oxygen)
        if event is GameEvent.BURNED:
            return self._die(event)
        if event is GameEvent.LEVEL_WON:
            return self._win()
        return [] if event is None else [event]

    def _die(self, cause: GameEvent) -> list[GameEvent]:
        self.phase = GamePhase.DYING
        self._phase_timer = self.config.dying_seconds
        return [cause]

    def _win(self) -> list[GameEvent]:
        self.phase = GamePhase.LEVEL_WON
        self._phase_timer = self.config.level_won_seconds
        return [GameEvent.LEVEL_WON]

    def _finish_transition(self) -> list[GameEvent]:
        if self._test_level is not None:
            if self.phase is GamePhase.DYING:
                return self.ride(self._test_level)
            self.open_editor()
            return []
        if self.phase is GamePhase.DYING:
            return self._begin_level(self.level_index)
        if self.level_index + 1 < len(self.levels):
            return self._begin_level(self.level_index + 1)
        self.phase = GamePhase.VICTORY
        return [GameEvent.GAME_COMPLETED]

    def _begin_level(self, index: int) -> list[GameEvent]:
        self._load_level(index)
        self.phase = GamePhase.PLAYING
        return [GameEvent.LEVEL_STARTED]

    def _load_level(self, index: int) -> None:
        self.level_index = index
        self._load(self.levels[index])

    def _load(self, level: Level) -> None:
        track = level.track
        start_x, start_z = track.width / 2, self.config.ship_length / 2 + 0.1
        start_y = max(
            (tile.top for tile in (track.tile_at(column, 0) for column in range(track.COLUMNS)) if tile is not None),
            default=Tile.HEIGHT_UNIT,
        )
        self.ship = Ship.from_config(Vector3(start_x, start_y, start_z), self.config)
        self.ship.grounded = track.surface_height_at(start_x, start_z) == start_y
        self.fuel = Gauge(level.fuel_capacity)
        self.oxygen = Gauge(level.oxygen_seconds)
        self._resolver = CollisionResolver(track, self.config)
        self._accumulator = 0.0

    class CollisionResolver:
    """Moves the ship through the track one axis at a time and resolves collisions.

    The ship collides with solids: tile blocks and the pieces of tunnel shells. Lateral motion is blocked
    by walls. Forward motion into a wall is a crash when a large part of the ship's nose hits it, otherwise
    the ship is nudged sideways around the corner. Vertical motion lands the ship on tops and bumps it
    against ceilings. Dropping below the fall limit means the ship fell off the track.

    Oblique tiles are solids whose top is the slope under the ship, so climbing one is a series of small
    steps. Going down, a grounded ship sticks to the slope instead of hopping off it. A ship leaving a slope
    keeps the climb (or dive) it had along it, so the top of a ramp is a kicker. Cresting a ramp onto flatter
    ground keeps part of the climb lost, for a little hop. A ship rising slower than the ramp under it lands on it.

    A grounded ship whose centre is on the edge of its support, or past it, drifts sideways toward the hole
    until it falls in.
    """

    EPSILON = 1e-6

    def __init__(self, track: Track, config: GameConfig) -> None:
        """Resolve collisions against the solids of ``track``."""
        self._track = track
        self._config = config

    def move(self, ship: Ship, dt: float, *, took_off: bool = False) -> list[CollisionReport]:
        """Move the ship by its velocity for ``dt`` seconds and report what it hit.

        ``took_off`` tells the ship jumped off the ground this step, so the tile it leaves still has its effect.
        """
        reports = self._move_laterally(ship, dt)
        reports += self._move_forward(ship, dt)
        if any(report.outcome is CollisionOutcome.CRASHED for report in reports):
            return reports
        reports.extend(self._move_vertically(ship, dt, took_off=took_off))
        if ship.position.y < self._config.fall_limit_y:
            reports.append(CollisionReport(CollisionOutcome.FELL))
        return reports

    def _blockers(self, box: BoundingBox) -> list[Solid]:
        """Solids in the way of ``box``: those that rise above a small step and start below its top."""
        floor = box.minimum.y + self._config.step_tolerance
        return [
            solid
            for solid in self._track.solids_under(box)
            if solid.top_over(box) > floor and solid.box.minimum.y < box.maximum.y - self.EPSILON
        ]

    def _move_laterally(self, ship: Ship, dt: float) -> list[CollisionReport]:
        """Strafes the ship, reporting a scrape on the side of any wall that stops it."""
        vx = ship.velocity.x
        if vx == 0:
            return []
        target = ship.position.with_x(ship.position.x + vx * dt)
        blockers = self._blockers(ship.box_at(target))
        if not blockers:
            ship.position = target
            return []
        ship.position = target.with_x(self._clear_x(ship, blockers, toward_positive=vx > 0))
        ship.velocity = ship.velocity.with_x(0.0)
        return [CollisionReport(CollisionOutcome.SCRAPED_RIGHT if vx > 0 else CollisionOutcome.SCRAPED_LEFT)]

    def _clear_x(self, ship: Ship, blockers: list[Solid], *, toward_positive: bool) -> float:
        """Return the x position that places the ship just beside ``blockers``."""
        half_width = ship.width / 2 + self.EPSILON
        if toward_positive:
            return min(solid.box.minimum.x for solid in blockers) - half_width
        return max(solid.box.maximum.x for solid in blockers) + half_width

    def _move_forward(self, ship: Ship, dt: float) -> list[CollisionReport]:
        target = ship.position.with_z(ship.position.z + ship.velocity.z * dt)
        box = ship.box_at(target)
        blockers = self._blockers(box)
        if not blockers:
            ship.position = target
            return []
        if self._covered_width(box, blockers) < self._config.crash_overlap_ratio * ship.width:
            nudged = self._nudge(ship, target, blockers)
            if nudged is not None:
                ship.position = nudged
                return [CollisionReport(CollisionOutcome.NUDGED, blockers[0].tile)]
        front = min(solid.box.minimum.z for solid in blockers)
        ship.position = ship.position.with_z(min(ship.position.z, front - ship.length / 2 - self.EPSILON))
        ship.velocity = ship.velocity.with_z(0.0)
        return [CollisionReport(CollisionOutcome.CRASHED, blockers[0].tile)]

    @staticmethod
    def _covered_width(box: BoundingBox, blockers: list[Solid]) -> float:
        """How much of the width of ``box`` the blockers cover, counting overlapping blockers once."""
        spans = sorted(
            (max(box.minimum.x, solid.box.minimum.x), min(box.maximum.x, solid.box.maximum.x)) for solid in blockers
        )
        covered, reach = 0.0, box.minimum.x
        for start, end in spans:
            if end > reach:
                covered += end - max(start, reach)
                reach = end
        return covered

    def _nudge(self, ship: Ship, target: Vector3, blockers: list[Solid]) -> Vector3 | None:
        """Slides the ship sideways off a corner, or returns None when there is no room to do so."""
        sides = {solid.box.center_x > target.x for solid in blockers}
        if len(sides) != 1:
            return None
        nudged = target.with_x(self._clear_x(ship, blockers, toward_positive=sides.pop()))
        return None if self._blockers(ship.box_at(nudged)) else nudged

    def _move_vertically(self, ship: Ship, dt: float, *, took_off: bool) -> list[CollisionReport]:
        old_y = ship.position.y
        new_y = old_y + ship.velocity.y * dt
        box = ship.box_at(ship.position.with_y(new_y))
        tops = [(solid, solid.top_over(box)) for solid in self._track.solids_under(box)]
        supports = [(solid, top) for solid, top in tops if top <= old_y + self._config.step_tolerance]
        surface = max((top for _, top in supports), default=None)
        if new_y > old_y and (surface is None or new_y > surface):
            takeoff = self._take_off(ship, supports, surface) if took_off else []
            return [*takeoff, *self._move_up(ship, old_y, new_y)]
        if surface is None or new_y > surface + (self._config.step_tolerance if ship.grounded else 0.0):
            if ship.grounded:
                across, along = ship.ground_slope
                climb = ship.velocity.x * across + ship.velocity.z * along
                ship.velocity = ship.velocity.with_y(ship.velocity.y + climb)
            ship.position = ship.position.with_y(new_y)
            ship.grounded = False
            ship.ground_slope = (0.0, 0.0)
            return []
        tile = self._tile_beneath(ship, supports, surface)
        slope = self._slope_under(ship, tile)
        kick = self._crest_kick(ship, slope) if ship.grounded else 0.0
        ship.position = ship.position.with_y(surface)
        ship.velocity = ship.velocity.with_y(kick)
        ship.grounded = kick == 0.0
        ship.ground_slope = slope if ship.grounded else (0.0, 0.0)
        self._drift_off_edge(ship, [solid for solid, _ in supports], dt)
        return [CollisionReport(CollisionOutcome.LANDED, tile)]

    def _crest_kick(self, ship: Ship, slope: tuple[float, float]) -> float:
        """Return the upward speed a ship climbing onto the flatter ``slope`` keeps, or 0 when too small to hop."""
        velocity = ship.velocity
        was_climbing = velocity.x * ship.ground_slope[0] + velocity.z * ship.ground_slope[1]
        climbing = velocity.x * slope[0] + velocity.z * slope[1]
        kick = self._config.crest_kick_ratio * (was_climbing - max(climbing, 0.0))
        return kick if kick >= self._config.min_crest_kick else 0.0

    def _drift_off_edge(self, ship: Ship, supports: list[Solid], dt: float) -> None:
        """Slides the ship toward the hole when its centre is on or past the side edge of its supports."""
        x, z = ship.position.x, ship.position.z
        beside = [solid for solid in supports if solid.box.minimum.z <= z <= solid.box.maximum.z]
        held_left = any(solid.box.minimum.x < x for solid in beside)
        held_right = any(solid.box.maximum.x > x for solid in beside)
        if held_left == held_right:
            return
        direction = 1.0 if held_left else -1.0
        target = ship.position.with_x(x + direction * self._config.edge_drift_speed * dt)
        if not self._blockers(ship.box_at(target)):
            ship.position = target

    def _take_off(
        self, ship: Ship, supports: list[tuple[Solid, float]], surface: float | None
    ) -> list[CollisionReport]:
        """Report the tile a ship jumps from, so its effect still applies: a jump pad still launches it."""
        if surface is None:
            return []
        return [CollisionReport(CollisionOutcome.LANDED, self._tile_beneath(ship, supports, surface))]

    def _move_up(self, ship: Ship, old_y: float, new_y: float) -> list[CollisionReport]:
        """Rises toward ``new_y``, stopping under any ceiling in the way."""
        ship.grounded = False
        ship.ground_slope = (0.0, 0.0)
        old_top = old_y + ship.height
        box = ship.box_at(ship.position.with_y(new_y))
        ceilings = [
            solid
            for solid in self._track.solids_under(box)
            if old_top - self.EPSILON <= solid.box.minimum.y < box.maximum.y
        ]
        if not ceilings:
            ship.position = ship.position.with_y(new_y)
            return []
        ceiling = min(solid.box.minimum.y for solid in ceilings)
        ship.position = ship.position.with_y(max(old_y, ceiling - ship.height - self.EPSILON))
        ship.velocity = ship.velocity.with_y(0.0)
        return [CollisionReport(CollisionOutcome.BUMPED)]

    def _tile_beneath(self, ship: Ship, supports: list[tuple[Solid, float]], surface: float) -> Tile | None:
        """Return the tile of the highest solid under the ship's centre, else of any highest support.

        None when the ship rests on a tunnel's shell, which has no effect.
        """
        highest = [solid for solid, top in supports if top == surface]
        position = ship.position
        for solid in highest:
            minimum, maximum = solid.box.minimum, solid.box.maximum
            if minimum.x <= position.x < maximum.x and minimum.z <= position.z < maximum.z:
                return solid.tile
        return highest[0].tile

    @staticmethod
    def _slope_under(ship: Ship, tile: Tile | None) -> tuple[float, float]:
        """Return how steeply ``tile`` rises under the ship, as height gained per unit along x and along z."""
        return (0.0, 0.0) if tile is None else tile.gradient(ship.position.x, ship.position.z)

    class ShipPhysics:
    """Turns player input, gravity and the surface underneath into the ship's velocity for one step.

    On ice the ship only slowly gets the sideways speed the player asks for, so it keeps sliding. Pushers
    add a constant sideways drift. Holding jump on the ground keeps hopping, and a jump off a ramp keeps the
    ramp's climb; once airborne, a fresh press gives one more jump, recharged on landing. That press does
    nothing while the ship already climbs faster than it would, so it never cuts short a jump pad's launch.
    """

    PUSH_DIRECTIONS: ClassVar[dict[TileType, int]] = {TileType.PUSH_LEFT: -1, TileType.PUSH_RIGHT: 1}

    def __init__(self, config: GameConfig) -> None:
        """Use the tunables from ``config``."""
        self._config = config

    def apply(self, ship: Ship, ship_input: ShipInput, gravity: float, dt: float) -> GameEvent | None:
        """Update ``ship.velocity`` and return the jump the ship made during this step, if any."""
        speed = self._forward_speed(ship.speed, ship_input, dt)
        lateral = self._lateral_speed(ship, speed, ship_input, dt)
        vertical = ship.velocity.y
        across, along = ship.ground_slope
        climb = lateral * across + speed * along
        jump = self._jump(ship, ship_input, rising=vertical)
        if jump is GameEvent.JUMPED:
            vertical = self._config.jump_velocity + climb
        elif jump is GameEvent.DOUBLE_JUMPED:
            vertical = self._config.double_jump_velocity
        ship.velocity = Vector3(lateral, vertical - gravity * dt, speed)
        return jump

    def _jump(self, ship: Ship, ship_input: ShipInput, rising: float) -> GameEvent | None:
        fresh_press = ship_input.jump and not ship.jump_held
        ship.jump_held = ship_input.jump
        if ship.grounded:
            ship.air_jump_ready = True
            if ship_input.jump:
                ship.grounded = False
                return GameEvent.JUMPED
        elif fresh_press and ship.air_jump_ready and rising < self._config.double_jump_velocity:
            ship.air_jump_ready = False
            return GameEvent.DOUBLE_JUMPED
        return None

    def _forward_speed(self, speed: float, ship_input: ShipInput, dt: float) -> float:
        config = self._config
        if speed > config.max_speed:
            speed = max(config.max_speed, speed - config.boost_decay * dt)
        elif ship_input.accelerate:
            speed = min(config.max_speed, speed + config.acceleration * dt)
        if ship_input.brake:
            speed -= config.brake_deceleration * dt
        return max(0.0, speed)

    def _lateral_speed(self, ship: Ship, speed: float, ship_input: ShipInput, dt: float) -> float:
        config = self._config
        direction = int(ship_input.right) - int(ship_input.left)
        wanted = direction * max(config.min_strafe_speed, speed * config.strafe_ratio)
        if ship.surface is TileType.ICE:
            current = ship.velocity.x
            return current + (wanted - current) * min(1.0, config.ice_grip * dt)
        return wanted + self.PUSH_DIRECTIONS.get(ship.surface, 0) * config.push_speed

    class TileEffectApplier:
    """Applies the effect of the tile the ship is riding on.

    It runs on every step the ship touches a tile, before ``ship.surface`` is updated, so ``ship.surface``
    still tells which tile the ship rode on before. Effects keep acting while the ship rides a patch, but
    the ones that would otherwise fire on every step (slowing, refills, ice, pushers) only report an
    event when the ship gets onto it.
    """

    RIDING_EVENTS: ClassVar[dict[TileType, GameEvent]] = {
        TileType.ICE: GameEvent.SKIDDED,
        TileType.PUSH_LEFT: GameEvent.PUSHED,
        TileType.PUSH_RIGHT: GameEvent.PUSHED,
    }
    ONCE_PER_PATCH = frozenset({GameEvent.SLOWED, GameEvent.REFUELED, GameEvent.REFILLED_AIR, *RIDING_EVENTS.values()})

    def __init__(self, config: GameConfig) -> None:
        """Use the tunables from ``config``."""
        self._config = config

    def apply(self, tile: Tile, ship: Ship, fuel: Gauge, oxygen: Gauge) -> GameEvent | None:
        """Return the event produced by the effect, or None when nothing worth reporting happened."""
        event = self._effect(tile, ship, fuel, oxygen)
        if event in self.ONCE_PER_PATCH and ship.surface is tile.tile_type:
            return None
        return event

    def _effect(self, tile: Tile, ship: Ship, fuel: Gauge, oxygen: Gauge) -> GameEvent | None:  # noqa: PLR0911
        match tile.tile_type:
            case TileType.BOOST:
                return self._boost(ship)
            case TileType.SLOW:
                return self._slow(ship)
            case TileType.KILL:
                return GameEvent.BURNED
            case TileType.REFUEL:
                return self._refill(fuel, GameEvent.REFUELED)
            case TileType.AIR:
                return self._refill(oxygen, GameEvent.REFILLED_AIR)
            case TileType.JUMP:
                return self._launch(ship)
            case TileType.FINISH:
                return GameEvent.LEVEL_WON
            case tile_type if tile_type in self.RIDING_EVENTS:
                return self.RIDING_EVENTS[tile_type]
            case _:
                return None

    def _boost(self, ship: Ship) -> GameEvent | None:
        boosted = self._config.max_speed * self._config.boost_factor
        if ship.speed >= boosted:
            return None
        was_boosting = ship.speed > self._config.max_speed
        ship.velocity = ship.velocity.with_z(boosted)
        return None if was_boosting else GameEvent.BOOSTED

    def _slow(self, ship: Ship) -> GameEvent | None:
        limit = self._config.max_speed * self._config.slow_factor
        if ship.speed <= limit:
            return None
        ship.velocity = ship.velocity.with_z(limit)
        return GameEvent.SLOWED

    def _launch(self, ship: Ship) -> GameEvent:
        ship.velocity = ship.velocity.with_y(self._config.launch_velocity)
        ship.grounded = False
        return GameEvent.LAUNCHED

    @staticmethod
    def _refill(gauge: Gauge, event: GameEvent) -> GameEvent | None:
        if gauge.is_full:
            return None
        gauge.refill()
        return event

    class LevelParser:
    """Parses the JSON level format.

    A level is a JSON object with an optional ``name``, ``gravity``, ``oxygen`` and ``fuel``, and a ``track``
    list, from the start of the track. Each entry is a ``[count, row]`` pair repeating ``row`` ``count`` times, or
    a lone ``row`` string for a single row; rows are written as :class:`TrackParser` reads them. ``#`` starts a
    comment, so a string starting with ``#`` is a comment line, and blank strings are skipped.
    """

    LEVEL = "level"
    TRACK = "track"
    COMMENT = "#"
    NUMERIC_KEYS: ClassVar[dict[str, str]] = {"gravity": "gravity", "oxygen": "oxygen_seconds", "fuel": "fuel_capacity"}

    def __init__(self, track_parser: TrackParser | None = None) -> None:
        """Parse tracks with ``track_parser``."""
        self._track_parser = track_parser or TrackParser()

    def parse(self, text: str) -> Level:
        """Parse the text of a level file."""
        try:
            document = json.loads(text)
        except json.JSONDecodeError as error:
            location, message = f"line {error.lineno}", f"invalid JSON: {error.msg}"
            raise LevelFormatError(location, message) from None
        if not isinstance(document, dict):
            raise LevelFormatError(self.LEVEL, "a level must be a JSON object")
        settings = self._settings(document)
        if self.TRACK not in document:
            raise LevelFormatError(self.LEVEL, f"missing {self.TRACK!r}")
        rows, locations = self._rows(document[self.TRACK])
        try:
            track = self._track_parser.parse(rows, locations)
        except TrackFormatError as error:
            raise LevelFormatError(error.location, str(error).removeprefix(f"{error.location}: ")) from None
        return Level(track=track, **settings)

    def _settings(self, document: dict) -> dict:
        """Return the level's settings, as :class:`Level` arguments, from every key but the track."""
        settings: dict = {"name": "Untitled"}
        for key, value in document.items():
            if key == "name":
                if not isinstance(value, str):
                    raise LevelFormatError(key, f"must be a string, got {value!r}")
                settings["name"] = value
            elif key in self.NUMERIC_KEYS:
                if isinstance(value, bool) or not isinstance(value, (int, float)):
                    raise LevelFormatError(key, f"must be a number, got {value!r}")
                settings[self.NUMERIC_KEYS[key]] = float(value)
            elif key != self.TRACK:
                raise LevelFormatError(key, "unknown key")
        return settings

    def _rows(self, track: object) -> tuple[list[TrackRow], list[str]]:
        """Return the track's rows as ``(count, row)`` pairs, comments left out, and where each one came from."""
        if not isinstance(track, list):
            raise LevelFormatError(self.TRACK, "must be a list of rows")
        rows: list[TrackRow] = []
        locations: list[str] = []
        for index, entry in enumerate(track):
            location = f"{self.TRACK}[{index}]"
            if isinstance(entry, str):
                count, line = 1, entry
            elif isinstance(entry, list) and len(entry) == 2:
                count, line = entry
            else:
                raise LevelFormatError(location, f"expected a row or a [count, row] pair, got {entry!r}")
            if isinstance(line, str):
                line = line.split(self.COMMENT, 1)[0].strip()
                if not line and isinstance(entry, str):
                    continue
            rows.append((count, line))
            locations.append(location)
        return rows, locations

    class TrackParser:
    """Parses track rows into a :class:`Track`.

    A track is a sequence of ``(count, row)`` pairs: ``row`` is repeated ``count`` times in a row, so a long
    straight is written once. A row is eight whitespace-separated cells: ``..`` for a gap, or a tile letter
    followed by a height digit (``N1``, ``B2``...), optionally followed by either a tunnel letter (``N1T`` for a
    square tunnel, ``N1A`` for a round one) or a slope naming the raised side or corner (``N1f`` rises toward the
    front, ``N1b``, ``N1l`` and ``N1r`` toward the back, left and right, ``N1fl``, ``N1fr``, ``N1bl`` and ``N1br``
    toward one corner). The height digit of a slope is its low side. The first row is the start of the track.
    """

    TRACK = "track"
    GAP = ".."
    MAX_LEVEL = 8
    TUNNELS = frozenset(kind.value for kind in TunnelKind)
    SLOPES = frozenset(slope.value for slope in Slope)

    def parse(self, rows: Sequence[TrackRow], locations: Sequence[str] | None = None) -> Track:
        """Expand and parse ``rows``. Errors name a row by its entry in ``locations``, or as ``row[index]``."""
        grid: list[list[Tile | None]] = []
        for index, entry in enumerate(rows):
            location = locations[index] if locations is not None else f"row[{index}]"
            if not isinstance(entry, tuple) or len(entry) != 2:
                raise TrackFormatError(location, f"expected a (count, row) pair, got {entry!r}")
            count, line = entry
            if isinstance(count, bool) or not isinstance(count, int) or count < 1:
                raise TrackFormatError(location, f"the count must be a positive integer, got {count!r}")
            if not isinstance(line, str):
                raise TrackFormatError(location, f"a row must be a string, got {line!r}")
            cells = line.split()
            if len(cells) != Track.COLUMNS:
                raise TrackFormatError(location, f"expected {Track.COLUMNS} cells, got {len(cells)}")
            for _ in range(count):
                row = len(grid)
                grid.append([self._parse_cell(location, row, column, cell) for column, cell in enumerate(cells)])
        if not grid:
            raise TrackFormatError(self.TRACK, "the track has no rows")
        return Track(grid)

    def _parse_cell(self, location: str, row: int, column: int, cell: str) -> Tile | None:
        if cell == self.GAP:
            return None
        if len(cell) < 2 or not cell[1].isdigit() or not 1 <= int(cell[1]) <= self.MAX_LEVEL:
            raise TrackFormatError(location, f"invalid cell {cell!r}")
        try:
            tile_type = TileType(cell[0])
        except ValueError:
            raise TrackFormatError(location, f"unknown tile type {cell[0]!r}") from None
        suffix = cell[2:]
        tunnel = slope = None
        if suffix in self.TUNNELS:
            tunnel = TunnelKind(suffix)
        elif suffix in self.SLOPES:
            slope = Slope(suffix)
        elif suffix:
            raise TrackFormatError(location, f"unknown tunnel or slope {suffix!r}")
        return Tile(tile_type, column, row, int(cell[1]), tunnel, slope)

    class Track:
    """A grid of tiles, eight columns wide, laid out row after row along the z axis.

    Side-by-side tunnel tiles of one kind and height in a row merge into a single tunnel span.
    """

    COLUMNS = 8

    def __init__(self, rows: Sequence[Sequence[Tile | None]]) -> None:
        """Build a track from ``rows`` of eight cells each, None meaning a gap."""
        if any(len(row) != self.COLUMNS for row in rows):
            msg = f"every track row must have {self.COLUMNS} cells"
            raise ValueError(msg)
        self._rows = tuple(tuple(row) for row in rows)
        self._tunnels = tuple(span for row in self._rows for span in self._spans_in(row))
        self._tunnels_by_row: dict[int, list[TunnelSpan]] = {}
        for span in self._tunnels:
            self._tunnels_by_row.setdefault(span.row, []).append(span)
        self._solids: dict[tuple[int, int], list[Solid]] = {}
        for tile in self.tiles():
            self._solids.setdefault((tile.column, tile.row), []).append(Solid(tile.bounding_box, tile))
        for span in self._tunnels:
            for solid in span.solids():
                first, last = math.floor(solid.box.minimum.x / Tile.WIDTH), math.ceil(solid.box.maximum.x / Tile.WIDTH)
                for column in range(first, last):
                    self._solids.setdefault((column, span.row), []).append(solid)
        # Each cell's solids with their footprints, as looked up by solids_under for every step of the ship.
        self._footprints = {
            cell: tuple(
                (solid, solid.box.minimum.x, solid.box.maximum.x, solid.box.minimum.z, solid.box.maximum.z)
                for solid in solids
            )
            for cell, solids in self._solids.items()
        }

    @staticmethod
    def _spans_in(row: Sequence[Tile | None]) -> Iterator[TunnelSpan]:
        run: list[tuple[TunnelKind, Tile]] = []
        for tile in (*row, None):
            if run and (tile is None or (tile.tunnel, tile.level) != (run[0][0], run[0][1].level)):
                kind, first = run[0]
                yield TunnelSpan(kind, first.row, first.column, run[-1][1].column, first.top)
                run = []
            if tile is not None and tile.tunnel is not None:
                run.append((tile.tunnel, tile))

    @property
    def row_count(self) -> int:
        """The number of rows."""
        return len(self._rows)

    @property
    def length(self) -> float:
        """The length of the track along the z axis."""
        return self.row_count * Tile.LENGTH

    @property
    def width(self) -> float:
        """The width of the track along the x axis."""
        return self.COLUMNS * Tile.WIDTH

    @property
    def tunnels(self) -> tuple[TunnelSpan, ...]:
        """Every tunnel span, row by row."""
        return self._tunnels

    def tile_at(self, column: int, row: int) -> Tile | None:
        """Return the tile in a cell, or None for a gap or a cell outside the track."""
        if 0 <= column < self.COLUMNS and 0 <= row < self.row_count:
            return self._rows[row][column]
        return None

    def tiles(self) -> Iterator[Tile]:
        """Yield every tile, row by row."""
        for row in self._rows:
            yield from (tile for tile in row if tile is not None)

    def tiles_under(self, box: BoundingBox) -> list[Tile]:
        """Tiles whose footprint overlaps the footprint of ``box``."""
        return [solid.tile for solid in self.solids_under(box) if solid.tile is not None]

    def solids_under(self, box: BoundingBox) -> list[Solid]:
        """Tile blocks and tunnel pieces whose footprint overlaps the footprint of ``box``."""
        first_column = math.floor(box.minimum.x / Tile.WIDTH)
        last_column = math.ceil(box.maximum.x / Tile.WIDTH)
        first_row = math.floor(box.minimum.z / Tile.LENGTH)
        last_row = math.ceil(box.maximum.z / Tile.LENGTH)
        low_x, high_x, low_z, high_z = box.minimum.x, box.maximum.x, box.minimum.z, box.maximum.z
        found: list[Solid] = []
        seen: set[int] = set()
        for row in range(first_row, last_row):
            for column in range(first_column, last_column):
                for solid, solid_low_x, solid_high_x, solid_low_z, solid_high_z in self._footprints.get(
                    (column, row), ()
                ):
                    # The test of BoundingBox.overlaps_footprint, on footprints read once.
                    if not (
                        low_x < solid_high_x > solid_low_x < high_x > low_x
                        and low_z < solid_high_z > solid_low_z < high_z > low_z
                    ):
                        continue
                    if id(solid) not in seen:
                        seen.add(id(solid))
                        found.append(solid)
        return found

    def ceiling_over(self, x: float, first_z: float, last_z: float) -> float | None:
        """Return the lowest tunnel ceiling above ``x`` from ``first_z`` to ``last_z``, or None outside tunnels."""
        first_row, last_row = math.floor(first_z / Tile.LENGTH), math.floor(last_z / Tile.LENGTH)
        ceilings = [
            span.inner_height(x)
            for row in range(first_row, last_row + 1)
            for span in self._tunnels_by_row.get(row, ())
            if span.left <= x < span.right
        ]
        return min(ceilings, default=None)

    def surface_height_at(self, x: float, z: float) -> float | None:
        """Return the height of the tile's top face at ``(x, z)``, or None over a gap."""
        tile = self.tile_at(math.floor(x / Tile.WIDTH), math.floor(z / Tile.LENGTH))
        return None if tile is None else tile.height_at(x, z)

        dt = 0.0 if self._last_timestamp is None else (timestamp - self._last_timestamp) / 1000

        """Pause when the page is hidden: another tab, or the phone's screen off."""
        if self._bridge.document.visibilityState == "hidden":

    def pause(self) -> None:
        """Freeze the ride, if one is on, behind the pause screen, and release the keys held."""
        if self.paused or self.session.phase not in (GamePhase.PLAYING, GamePhase.DYING):
            return

        self._keyboard.release_all()
        self._music.pause()

    def resume(self) -> None:
        """Go on with a paused ride."""
        if self.paused:
            self.paused = False
            self._music.resume()

        """Quit a paused ride, back to the editor it was started from or to the level select."""
        self.resume()

        else:

    def tick(self, dt: float) -> None:
        """Advance the whole game by ``dt`` seconds and render a frame, or only redraw while paused."""
        session = self.session
        for key in self._keyboard.consume_presses():
            self.handle_key(key)
        for tap in self._menu.consume_taps():
            self.handle_tap(tap)
        if self.paused:
            self._update_overlays()
            self._renderer.render()
            return
        self._time += dt
        self._handle_editor_actions()
        events = session.update(dt, self._keyboard.ship_input())
        self._sound_effects.play(events)
        self._renderer.react(events)
        self._sparks.react(events, session.ship)
        self._sparks.update(dt)
        self._hud.announce(events)
        self._hud.fade(dt)
        self._hud.count_frame(dt)
        self._music.update()

        snap = self._show_level(session.level if editor is None else editor.level)
        snap = snap or GameEvent.LEVEL_STARTED in events
        self._ship_model.sync(session.ship, session.phase, dt)
        self._ship_model.group.visible = editor is None
        if editor is not None:
            editor.update(dt)
            self._renderer.overlook(*editor.focus)
        elif session.phase is not GamePhase.DYING:
            ship = session.ship
            speed_fraction = ship.speed / session.config.max_speed
            ceiling = session.level.track.ceiling_over(
                ship.position.x, ship.position.z - self.CAMERA_REACH_BEHIND, ship.position.z + self.CAMERA_REACH_AHEAD
            )
            self._renderer.follow(ship, min(speed_fraction, 1.3), dt, snap=snap, ceiling=ceiling)
            self._renderer.reflect_on(ship, session.level.track.surface_height_at(ship.position.x, ship.position.z))
        self._backdrop.update(self._renderer.camera_z, self._time)
        self._track_builder.animate(self._time)
        self._update_overlays()
        self._renderer.render()

    def handle_key(self, key: str) -> None:
        """React to a fresh key press in the menus. The first one starts the music: browsers wait for a gesture."""
        session, phase = self.session, self.session.phase
        self._start_music()
        letter = self._keyboard.letter(key)
        if letter == self.MUTE:
            self._music.toggle_mute()
        elif self.paused:
            self._handle_pause_key(key)
        elif phase is GamePhase.EDITOR and self._editor is not None:
            self._editor.handle_key(key, letter)
        elif letter in (self.EDIT, self.NEW) and phase is GamePhase.LEVEL_SELECT and self._editor is not None:
            self._open_editor(self._editor, selected=letter == self.EDIT)
        elif phase in (GamePhase.PLAYING, GamePhase.DYING) and key == "Escape":
            self.pause()
        elif phase is GamePhase.LEVEL_SELECT:
            self._handle_level_select_key(key)
        elif phase is GamePhase.VICTORY and key in self.CONFIRM:
            session.open_level_select()

    def _start_music(self) -> None:
        if not self._music_started:
            self._music_started = True
            self._music.start()

    def _handle_pause_key(self, key: str) -> None:
        if key in ("Escape", "Enter"):
            self.resume()
        elif key == "Backspace":
            self.leave()

    def handle_tap(self, tap: int | str) -> None:
        """React to a tap on the menus, as the key it stands for: a level rides it, back is Escape, else Enter."""
        session, phase = self.session, self.session.phase
        self._start_music()
        if self.paused:
            if tap == self._menu.RESUME:
                self.resume()
            elif tap == self._menu.MAIN_MENU:
                self.leave()
        elif phase is GamePhase.LEVEL_SELECT:
            if isinstance(tap, int) and 0 <= tap < len(session.levels):
                session.selected_level = tap
                self.handle_key("Enter")
        elif phase in self.TAP_TO_GO_ON and tap == self._menu.ANYWHERE:
            self.handle_key("Enter")

    def _open_editor(self, editor: EditorScreen, *, selected: bool) -> None:
        """Open the editor on the level selected in the level select, or else on the last track it had."""
        session = self.session
        if selected:
            index = session.selected_level
            editor.open_level(session.levels[index], index)
        session.open_editor()

    def _handle_editor_actions(self) -> None:
        if self._editor is None:
            return
        session = self.session
        for action in self._editor.consume_actions():
            if session.phase is not GamePhase.EDITOR:
                continue
            if action == self._editor.RIDE:
                self._sound_effects.play(session.ride(self._editor.level))
            else:  # BACK
                if self._editor.index is not None:
                    session.selected_level = self._editor.index
                session.open_level_select()

    def _handle_level_select_key(self, key: str) -> None:
        session = self.session
        if key in self.UP:
            session.select_previous_level()
        elif key in self.DOWN:
            session.select_next_level()
        elif key in self.LEFT:
            session.select_level_column(-1, self._menu.LEVELS_PER_COLUMN)
        elif key in self.RIGHT:
            session.select_level_column(1, self._menu.LEVELS_PER_COLUMN)
        elif key in self.CONFIRM:
            self._sound_effects.play(session.start())

    def _show_level(self, level: Level) -> bool:
        """Rebuilds the track meshes and redresses the sky when ``level`` is not the one shown. Returns True if so."""
        if level is self._shown_level:
            return False
        if self._track_group is not None:
            self._renderer.remove(self._track_group)
        self._track_group = self._track_builder.build(level.track)
        self._renderer.add(self._track_group)
        self._backdrop.show(SpaceScenery.for_level(self.session.level_index))
        self._shown_level = level
        return True

    def _update_overlays(self) -> None:
        session, phase = self.session, self.session.phase
        self._hud.show(visible=phase in self.HUD_PHASES)
        if phase in self.HUD_PHASES:
            self._hud.update(session)
        if self.paused:
            self._menu.show_pause(riding=session.riding)
            return
        match phase:
            case GamePhase.LEVEL_SELECT:
                names = [level.name for level in session.levels]
                self._menu.show_level_select(names, session.selected_level, editor=self._editor is not None)
            case GamePhase.LEVEL_WON:
                self._menu.show_level_won(session.level.name)
            case GamePhase.VICTORY:
                self._menu.show_victory()
            case _:
                self._menu.hide()
        if self._editor is not None:
            if phase is GamePhase.EDITOR:
                self._editor.show()
            else:
                self._editor.hide()

    def _vertical_fov(self, fov: float) -> float:
        """Return ``fov``, widened for a window taller than it is wide so the track keeps room at the sides.

        The field of view is vertical: held upright the view would shrink to a slit. The horizontal angle is kept
        halfway, in tangent terms, between the landscape's and the square's.
        """
        if self.camera.aspect >= 1:
            return fov
        return math.degrees(2 * math.atan(math.tan(math.radians(fov) / 2) / math.sqrt(self.camera.aspect)))

            self._shake = 0.5 if hard else 0.15

    def follow(
        self, ship: Ship, speed_fraction: float, dt: float, *, snap: bool = False, ceiling: float | None = None
    ) -> None:
        """Eases the chase camera toward its spot behind the ship and widens the view with speed.

        Under a ``ceiling`` (a tunnel roof), the camera drops below it so the roof never hides the ship. While the
        ship is airborne the camera stays level with the ground it left, so a fall never shows the track from under.
        """
        x, y, z = ship.position.x, ship.position.y, -ship.position.z
        if ship.grounded or snap:
            self._ground = y
        height = max(y, self._ground) + self.CAMERA_HEIGHT
        if ceiling is not None:
            height = max(y + self.MIN_HEIGHT, min(height, ceiling - self.CEILING_MARGIN))
        target = (x, height, z + self.CAMERA_DISTANCE)
        blend = 1.0 if snap else 1 - math.exp(-self.FOLLOW_RATE * dt)
        self._camera_position = [
            current + (goal - current) * blend for current, goal in zip(self._camera_position, target, strict=True)
        ]
        self._camera_position[2] = target[2]  # never lag behind along the track
        self._shake *= math.exp(-6 * dt)
        jitter = [self._rng.uniform(-self._shake, self._shake) for _ in range(2)]
        cx, cy, cz = self._camera_position
        self._eye = (cx + jitter[0], cy + jitter[1], cz)
        self._look = (x, y + 0.3, z - self.LOOK_AHEAD)
        self.camera.position.set(*self._eye)
        self.camera.lookAt(*self._look)
        self.camera.fov = self._vertical_fov(self.BASE_FOV + self.SPEED_FOV * speed_fraction)
        self.camera.updateProjectionMatrix()

    def overlook(self, x: float, z: float, ground: float) -> None:
        """Place the editor's camera raised behind the point ``(x, z)`` at ``ground`` height, looking down the track."""
        self._eye = (x, ground + self.EDITOR_HEIGHT, z + self.EDITOR_DISTANCE)
        self._look = (x, ground, z - self.EDITOR_LOOK_AHEAD)
        self._camera_position = list(self._eye)
        self.camera.position.set(*self._eye)
        self.camera.lookAt(*self._look)
        self.camera.fov = self._vertical_fov(self.BASE_FOV)
        self.camera.updateProjectionMatrix()

    def reflect_on(self, ship: Ship, deck: float | None) -> None:
        """Mirror ``ship`` in the deck at height ``deck`` under it (None over a gap)."""
        self.reflection.place((ship.position.x, ship.position.y, -ship.position.z), deck)

    def render(self) -> None:
        """Render a frame: the ship's mirror image first, then the scene through post-processing."""
        self.reflection.render(self.renderer, self.scene, self.camera, self._eye, self._look)
        self.composer.render()  # each pass binds its own target, the mirror's included

    class BundleBuilder:
    """Zips the client side of the game (core, editor, web and levels) for Pyodide to unpack in the browser."""

    CLIENT_PACKAGES = ("core", "editor", "web", "levels")
    SUFFIXES = frozenset({".py", ".json"})
    TIMESTAMP = (2024, 1, 1, 0, 0, 0)

    def __init__(self, package_root: Path | None = None) -> None:
        """Bundle the package found at ``package_root``, by default this one."""
        self._package_root = package_root or Path(__file__).resolve().parent.parent

    def files(self) -> list[Path]:
        """Return the paths to bundle, relative to the directory that contains the package."""
        root = self._package_root
        candidates = [root / "__init__.py"]
        for package in self.CLIENT_PACKAGES:
            candidates.extend(sorted((root / package).rglob("*")))
        return [
            path.relative_to(root.parent)
            for path in candidates
            if path.is_file() and path.suffix in self.SUFFIXES and "__pycache__" not in path.parts
        ]

    def entries(self) -> list[tuple[str, bytes]]:
        """Return the archive's ``(name, content)`` pairs."""
        return [(relative.as_posix(), (self._package_root.parent / relative).read_bytes()) for relative in self.files()]

    def build(self) -> bytes:
        """Return the zip archive of the client files."""
        buffer = io.BytesIO()
        with zipfile.ZipFile(buffer, "w", zipfile.ZIP_DEFLATED) as archive:
            for name, content in self.entries():
                info = zipfile.ZipInfo(name, self.TIMESTAMP)
                info.compress_type = zipfile.ZIP_DEFLATED
                archive.writestr(info, content)
        return buffer.getvalue()

    def __init__(
        self,
        pyodide_version: str = PYODIDE_VERSION,
        three_version: str = THREE_VERSION,
        bundle_url: str = "synthride.zip",
        *,
        dev: bool = False,
        commit: str | None = None,
    ) -> None:
        """Use the given library versions and load the bundle from ``bundle_url``; ``dev`` turns the editor on.

        The menu shows ``commit``, by default the one the working tree is on.
        """
        self._dev = dev
        self._commit = commit or self.current_commit()
        self._pyodide_version = pyodide_version
        self._three_version = three_version
        self._bundle_url = bundle_url

    @staticmethod
    def current_commit() -> str:
        """Return the short SHA of the commit being built: ``GITHUB_SHA`` in CI, else git's HEAD, else ``unknown``."""
        sha = os.environ.get("GITHUB_SHA")
        if sha:
            return sha[:7]
        try:
            result = subprocess.run(
                ["git", "rev-parse", "--short", "HEAD"],  # noqa: S607
                capture_output=True,
                text=True,
                check=True,
                cwd=Path(__file__).parent,
            )
        except (OSError, subprocess.CalledProcessError):
            return "unknown"
        return result.stdout.strip() or "unknown"

    @property
    def pyodide_url(self) -> str:
        """The CDN folder Pyodide is loaded from."""

    @staticmethod
def current_commit() -> str:
    """Return the short SHA of the commit being built: ``GITHUB_SHA`` in CI, else git's HEAD, else ``unknown``."""
    sha = os.environ.get("GITHUB_SHA")
    if sha:
        return sha[:7]
    try:
        result = subprocess.run(
            ["git", "rev-parse", "--short", "HEAD"],  # noqa: S607
            capture_output=True,
            text=True,
            check=True,
            cwd=Path(__file__).parent,
        )
    except (OSError, subprocess.CalledProcessError):
        return "unknown"
    return result.stdout.strip() or "unknown"

    """Writes level files sent by the editor into the levels package, checking each one first."""

    NAME: ClassVar[re.Pattern[str]] = re.compile(r"level_\d{2,}\.json")
    MAX_BYTES: ClassVar[int] = 2_000_000

    def __init__(self, root: Path | None = None) -> None:
        """Save into ``root``, by default the bundled levels package."""
        self.root = root or Path(__file__).resolve().parent.parent / "levels"

    def save(self, name: str, text: str) -> Path:
        """Write the level file ``text`` as ``name`` and return its path.

        Raise ``ValueError`` if ``name`` is not a level file name, such as ``level_07.json``, or ``text`` is not a
        level file.
        """
        if not self.NAME.fullmatch(name):
            msg = f"not a level file name: {name!r}"
            raise ValueError(msg)
        LevelParser().parse(text)
        path = self.root / name
        path.write_text(text if text.endswith("\n") else f"{text}\n", encoding="utf-8")
        return path
